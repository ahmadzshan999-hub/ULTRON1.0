<script>
let currentPlan = "Free";

function showPage(page){
  document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));
  document.getElementById(page).classList.add('active');
  const names={chat:'AI Assistant',pricing:'Subscriptions',account:'My Account'};
  document.getElementById('pageName').textContent=names[page];
}

function newChat(){
  document.getElementById('messages').innerHTML='';
  document.getElementById('welcome').style.display='block';
  showPage('chat');
}

function usePrompt(text){
  document.getElementById('messageInput').value=text;
  sendMessage();
}

function sendMessage(){
  const input=document.getElementById('messageInput');
  const value=input.value.trim();
  if(!value)return;

  document.getElementById('welcome').style.display='none';
  const messages=document.getElementById('messages');

  const user=document.createElement('div');
  user.className='message user';
  user.textContent=value;
  messages.appendChild(user);

  const reply=document.createElement('div');
  reply.className='message bot';
  reply.textContent="ULTRON is currently running in demo mode. Connect your OpenAI API through a secure backend to enable real AI conversations. Your interface is ready for that connection.";
  messages.appendChild(reply);

  input.value='';
  messages.scrollTop=messages.scrollHeight;
}

function selectPlan(plan){
  currentPlan=plan;
  document.getElementById('sidebarPlan').textContent=plan+' Plan';
  document.getElementById('accountStatus').textContent='Current plan: '+plan;
  alert('Your demo plan is now '+plan+'. This does not activate a paid subscription.');
}

function subscribe(plan){
  alert('The '+plan+' checkout is not connected yet. A secure payment backend and payment-provider setup are required.');
}

function saveAccount(){
  const name=document.getElementById('displayName').value.trim();
  const email=document.getElementById('email').value.trim();
  if(!name||!email){alert('Please enter your name and email.');return;}
  document.getElementById('accountStatus').textContent='Profile details entered for this demo. They are not securely saved.';
  alert('Demo profile updated.');
}
</script>
